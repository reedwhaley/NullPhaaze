cmake_minimum_required(VERSION 3.15)
project(nullphaaze_lzokay LANGUAGES CXX)
add_library(lzokay SHARED lzokay/lzokay.cpp bridge.cpp)
if(WIN32)
  target_sources(lzokay PRIVATE exports.def)
endif()
target_compile_features(lzokay PRIVATE cxx_std_14)
set_target_properties(lzokay PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  set_property(TARGET lzokay PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
